n = int(input("Enter size of array: "))
arr = list(map(int, input("Enter array elements: ").split()))

k = int(input("Enter k: "))

count = 0

for i in range(n):
    for j in range(i + 1, n):
        if arr[i] == arr[j] and (i * j) % k == 0:
            count += 1

print("Count:", count)
